<?php
session_start();

require_once "conexao.php";

$mensagem = '';
$tipoMensagem = '';

if ($_SERVER['REQUEST_METHOD'] === 'POST') {

    // Recebe os dados do formulário
    $nome = trim($_POST['name'] ?? '');
    $email = trim($_POST['email'] ?? '');
    $senha = $_POST['password'] ?? '';
    $telefone = trim($_POST['telefone'] ?? '');
    $cpf = trim($_POST['cpf'] ?? '');
    $endereco = trim($_POST['address'] ?? '');
    $cep = trim($_POST['cep'] ?? '');
    $numero = trim($_POST['address_number'] ?? '');
    $complemento = trim($_POST['complement'] ?? '');

    // ==========================
    // VALIDAÇÕES
    // ==========================

    if (empty($nome) || empty($email) || empty($senha)) {

        $mensagem = 'Preencha todos os campos obrigatórios.';
        $tipoMensagem = 'danger';

    } elseif (!filter_var($email, FILTER_VALIDATE_EMAIL)) {

        $mensagem = 'Digite um endereço de e-mail válido.';
        $tipoMensagem = 'danger';

    } elseif (strlen($senha) < 8) {

        $mensagem = 'A senha deve possuir pelo menos 8 caracteres.';
        $tipoMensagem = 'danger';

    } else {

        // Protege a senha antes de armazená-la
        $senhaHash = password_hash($senha, PASSWORD_DEFAULT);

        $dados = [
            'name'           => $nome,
            'email'          => $email,
            'password'       => $senhaHash,
            'telefone'       => $telefone,
            'cpf'            => $cpf,
            'address'        => $endereco,
            'cep'            => $cep,
            'address_number' => $numero,
            'complement'     => $complemento,
            'role'           => 'Cliente',
            'status'         => 'Ativo'
        ];

        try {

            $usuarioModel = new Usuario();

            if ($usuarioModel->cadastrar($dados)) {

                $mensagem = 'Registro efetuado com sucesso! 
                             <a href="login.php" class="alert-link">
                             Faça login aqui</a>.';

                $tipoMensagem = 'success';

                // Limpa os campos depois do cadastro
                $_POST = [];

            } else {

                $mensagem = 'Não foi possível realizar o cadastro. 
                             Verifique os dados e tente novamente.';

                $tipoMensagem = 'danger';
            }

        } catch (Exception $e) {

            $mensagem = 'Ocorreu um erro ao realizar o cadastro. 
                         Tente novamente mais tarde.';

            $tipoMensagem = 'danger';
        }
    }
}
