$erro = '';

if ($_SERVER['REQUEST_METHOD'] === 'POST') {

    $email = filter_input(INPUT_POST, 'email', FILTER_VALIDATE_EMAIL);

    if (!$email || empty($senha)) {
        $erro = "Preencha email e senha corretamente.";
    } else {

        $stmt = mysqli_prepare($conn, "SELECT id, senha, nivel, email FROM cadastro WHERE email = ?");
        mysqli_stmt_bind_param($stmt, "s", $email);
        mysqli_stmt_execute($stmt);
        $result = mysqli_stmt_get_result($stmt);
         if (mysqli_num_rows($result) === 1) {
            $usuario = mysqli_fetch_assoc($result);

            if (password_verify($senha, $usuario['senha'])) {

                // Login confirmado: cria a sessão
                session_regenerate_id(true); // gera um novo ID de sessão ao logar, evita "session fixation"
                $_SESSION['usuario_id'] = $usuario['id'];
                $_SESSION['email']      = $usuario['email'];
                $_SESSION['admin']      = ($usuario['nivel'] === 'admin');

                if ($_SESSION['admin']) {
                    header("Location: index.php"); // primeiro passa pela verificação do código por e-mail
                } else {
                    header("Location: area_cliente.php"); // ajuste para a página real do cliente
                }
                exit();

                $erro = "Email ou senha incorretos.";
            }
        } else {
            $erro = "Email ou senha incorretos.";
        }
    }
